\part*{Systems of First Order Linear ODEs}

\section{First Order ODE Systems}

\subsection{Non-Linear and Linear Systems}

The general system of first order equations involving
\(n\) dependent variables defined on an interval \(I \subseteq \mathbb{R}\)
can be written as
\begin{equation}
  \begin{gathered}
    \begin{cases}
      y_1'(t) = F_1(t, y_1, \ldots, y_n) \\
      y_2'(t) = F_2(t, y_1, \ldots, y_n) \\
      \qquad\vdots \\
      y_n'(t) = F_n(t, y_1, \ldots, y_n)
    \end{cases} \\
    \text{with initial conditions}\quad
    y_1(t_0) = x_1, \quad
    y_2(t_0) = x_2, \quad
    \ldots, \quad
    y_n(t_0) = x_n,
  \end{gathered}
  \qfor t\in I.
  \label{eq: system of 1st order ODEs}
\end{equation}
The solution to the system can be expressed in vector form as
\begin{equation*}
  \vec{y}(t) =
  \begin{pmatrix}
    y_1(t) \\
    y_2(t) \\
    \vdots \\
    y_n(t)
  \end{pmatrix}
\end{equation*}

\begin{theorem}[Existence and Uniqueness]
  {}\label{thm: Existence and Uniqueness for 1st order ODE Systems}
  For the system of first order ODEs~(\ref{eq: system of 1st order ODEs}),
  if the functions \(F_1, F_2, \ldots, F_n\) and the partial derivatives
  \(\displaystyle\pdv{F_i}{y_j}\) for all pairs \(i, j = 1, 2, \ldots, n\)
  are continuous on a region \(R\) defined by
  \begin{equation*}
    R := [t_0 - a, t_0 + a] \times [x_1 - b, x_1 + b] \times \cdots
    \times [x_n - b, x_n + b] \subseteq \mathbb{R}^{n+1},
  \end{equation*}
  where constants \(a, b > 0\), then there exists a unique solution
  \(\vec{y}(t)\) for \(t \in (t_0 - h, t_0 + h)\) to the IVP,
  where
  \begin{equation*}
    h = \min \{a, b/M\} \qand
    M = \max\limits_{\vec{v} \in R}\{
      \abs{F_1(\vec{v})}, \abs{F_2(\vec{v})}, \ldots, \abs{F_n(\vec{v})}
    \}.
  \end{equation*}
\end{theorem}

Any \(n\)-th order ODE can be rewritten as a system of \(n\) first order ODEs.
Given an \(n\)-th order ODE
\begin{equation*}
  y^{(n)} = F(t, y, y', \ldots, y^{(n-1)}),
\end{equation*}
setting \(z_1 = y, z_2 = y', \ldots, z_n = y^{(n-1)}\), we obtain the system
\begin{equation*}
  \begin{cases}
    z_1' = z_2 \\
    z_2' = z_3 \\
    \qquad\vdots \\
    z_{n-1}' = z_n \\
    z_n' = F(t, z_1, z_2, \ldots, z_n)
  \end{cases}
\end{equation*}
which is a system of \(n\) first order ODEs.

\begin{definition}[First-Order Linear System]
  {}\label{def: 1st-order linear system}
  A system of first order ODEs is called \textbf{linear} if it can be
  expressed in the form
  \begin{equation}
    \vec{y}'(t) = \mat{P}(t) \vec{y}(t) + \vec{g}(t) \quad \qfor t\in I,
    \label{eq: 1st-order linear system}
  \end{equation}
  where
  \begin{equation*}
    \vec{y}(t) =
    \begin{pmatrix}
      y_1(t) \\
      y_2(t) \\
      \vdots \\
      y_n(t)
    \end{pmatrix}
    \qquad
    \vec{g}(t) =
    \begin{pmatrix}
      g_1(t) \\
      g_2(t) \\
      \vdots \\
      g_n(t)
    \end{pmatrix}
    \qquad
    \mat{P}(t) =
    \begin{pmatrix}
      p_{11}(t) & p_{12}(t) & \cdots & p_{1n}(t) \\
      p_{21}(t) & p_{22}(t) & \cdots & p_{2n}(t) \\
      \vdots & \vdots & \ddots & \vdots \\
      p_{n1}(t) & p_{n2}(t) & \cdots & p_{nn}(t)
    \end{pmatrix}
  \end{equation*}
\end{definition}

\begin{definition}[Homogeneous First-Order Linear System]
  {}\label{def: Homogeneous 1st-order linear system}
  A first order linear system is called \textbf{homogeneous} if
  \(\vec{g}(t) = \vec{0}_n\) for all \(t\in I\), i.e., it can be expressed as
  \begin{equation*}
    \vec{y}'(t) = \mat{P}(t) \vec{y}(t) \quad \qfor t\in I.
  \end{equation*}
\end{definition}

\begin{theorem}[Existence and Uniqueness for First-Order Linear Systems]
  {}\label{thm: Existence and Uniqueness for 1st-order linear systems}
  For the first-order linear system~(\ref{eq: 1st-order linear system}),
  if all entries of \(\mat{P}(t)\) and \(\vec{g}(t)\) are continuous on
  an interval \(I\), then for any \(t_0 \in I\) and \(\vec{y}_0 \in
  \mathbb{R}^n\),
  there exists a unique solution \(\vec{y}(t)\) to the IVP
  \begin{equation*}
    \begin{cases}
      \vec{y}'(t) = \mat{P}(t) \vec{y}(t) + \vec{g}(t) \\
      \vec{y}(t_0) = \vec{a} \in \mathbb{R}^n
    \end{cases}
    \quad \qfor t\in I.
  \end{equation*}
\end{theorem}

\subsection{Homogeneous Linear Systems}

In this section,
we will focus on the following homogeneous
system defined on an open interval \(I \subseteq \mathbb{R}\):
\begin{equation}
  \vec{y}'(t) = \mat{P}(t) \vec{y}(t) \qfor t\in I,
  \label{eq: 1.HL.Sys}
\end{equation}
where \(\vec{y} \in {C^1(I)}^n\) and
\(\mat{P} \in {C(I)}^{n \times n}\),
i.e., all entries of \(\mat{P}\) and \(\vec{y}\) are continuous on \(I\).

\smallskip

\begin{theorem}[Principle of Superposition]
  {}\label{thm: Principle of Superposition}
  If \(\vec{y}_1(t)\) and \(\vec{y}_2(t)\) are solutions to the homogeneous
  first-order linear system~(\ref{eq: 1.HL.Sys}),
  then the linear combination
  \begin{equation*}
    \vec{y}(t) = c_1 \vec{y}_1(t) + c_2 \vec{y}_2(t)
  \end{equation*}
  is also a solution for any constants \(c_1, c_2\).
\end{theorem}

\begin{definition}[Wronskian]
  The \textbf{Wronskian} of \(n\) vector functions
  \(\vec{f}_1(t), \vec{f}_2(t), \ldots, \vec{f}_n(t) \in {C(I)}^n \)
  is defined as
  \begin{equation*}
    W(\vec{f}_1, \vec{f}_2, \ldots, \vec{f}_n)(t)
    \quad
    :=
    \quad
    \det(
      \begin{bmatrix}
        \vrule & & \vrule \\
        \vec{f}_1(t) & \cdots & \vec{f}_n(t) \\
        \vrule & & \vrule
    \end{bmatrix})
    \qfor t\in I.
  \end{equation*}
\end{definition}

\begin{remark}
  The Wronskian for vector functions is different from
  the Wronskian for scalar functions.
  There is no derivative involved in the definition of
  the Wronskian for vector functions.
\end{remark}

\begin{theoremBox}
  \noindent For any \( t \in I \),
  \begin{equation*}
    W(\vec{f}_1, \vec{f}_2, \ldots, \vec{f}_n)(t) \ne 0
    \quad\iff\quad
    \vec{f}_1(t), \vec{f}_2(t), \ldots, \vec{f}_n(t)
    \text{ are linearly independent}.
  \end{equation*}
\end{theoremBox}

\begin{theorem}[Liouville's Formula]
  {}\label{thm: Liouville's Formula}
  Let \(\vec{y}_1(t), \vec{y}_2(t), \ldots,
  \vec{y}_n(t)\) be \(n\) solutions
  to the homogeneous system~(\ref{eq: 1.HL.Sys}).
  Then the Wronskian of these solutions is given by
  \begin{equation*}
    W(\vec{y}_1, \vec{y}_2, \ldots, \vec{y}_n)(t) =
    \textcolor{red}{c} \cdot
    \exp( \int \tr(\mat{P}(t)) \dd{t} )
    \qfor t\in I,
  \end{equation*}
  where \(\textcolor{red}{c}\) is a constant
  not depending on \(t\).
\end{theorem}
\begin{remark}
  The \textbf{trace} of a square matrix
  \(\mat{A} \in \mathbb{R}^{n \times n}\)
  is defined as the sum of its diagonal entries:
  \begin{equation*}
    \tr(\mat{A}) := a_{11} + a_{22} + \cdots + a_{nn}.
  \end{equation*}
\end{remark}
\begin{proof}[Liouville's Formula]
  Before the proof, we need to introduce several theorems from linear algebra:
  \begin{enumerate}
    \item For any square matrix \(\mat{A}\),
      \begin{equation}
        \dv{t}(\det(\mat{A})) = \tr(\adj(\mat{A}) \dv{\mat{A}}{t}).
        \label{thm: local.01 dv of det}
      \end{equation}
    \item The trace remains unchanged under cycle permutations:
      \begin{equation}
        \tr(\mat{A}\mat{B}\mat{C}) = \tr(\mat{B}\mat{C}\mat{A}).
        \label{thm: local.02 trace cycle perm}
      \end{equation}
    \item Property of the adjugate matrix:
      \begin{equation}
        \adj(\mat{A}) \mat{A} = \mat{A} \adj(\mat{A})
        = \det(\mat{A}) \mat{I}.
        \label{thm: local.03 adj property}
      \end{equation}
  \end{enumerate}

  Now, let \(\mat{Y} :=
    \begin{bmatrix}
      \vec{y}_1 & \vec{y}_2 & \cdots & \vec{y}_n
  \end{bmatrix} \in {C^1(I)}^{n \times n}\)
  and \(W := \det(\mat{Y})\) be the Wronskian.
  Then we have
  \begin{align*}
    W' &= \tr(\adj(\mat{Y}) \mat{Y}') \\
    &= \tr(\adj(\mat{Y}) \mat{P} \mat{Y})
    \qquad\qsince \vec{y}' = \mat{P} \vec{y} \\
    &= \tr(\mat{Y} \adj(\mat{Y}) \mat{P}) \\
    &= \tr(W \mat{I} \mat{P}) \\
    &= W \tr(\mat{P}).
  \end{align*}
  Hence the Liouville's formula.
\end{proof}

\begin{theorem}
  {}\label{thm: nonzero Wronskian}
  If \(\vec{y}_1, \vec{y}_2, \ldots, \vec{y}_n\) are
  \(n\) solutions to
  a homogeneous system~(\ref{eq: 1.HL.Sys}),
  then
  \begin{equation*}
    \forall t\in I, \quad
    W(\vec{y}_1, \vec{y}_2, \ldots, \vec{y}_n)(t) = 0
    \qquad\iff\qquad
    \exists t_0 \in I \text{ s.t. }
    W(\vec{y}_1, \vec{y}_2, \ldots, \vec{y}_n)(t_0) = 0.
  \end{equation*}
\end{theorem}
\begin{proof}
  Proven by Liouville's formula.
\end{proof}

\begin{theorem}
  If the \(n\) solutions \(\vec{y}_1(t), \vec{y}_2(t),
  \ldots, \vec{y}_n(t)\)
  of the homogeneous system~(\ref{eq: 1.HL.Sys})
  are \textbf{linearly independent} for each \(t\in I\),
  then every solution of the system can be expressed as a linear combination:
  \begin{equation*}
    \vec{y}(t) = c_1 \vec{y}_1(t) + c_2 \vec{y}_2(t) + \cdots
    + c_n \vec{y}_n(t)
    \qfor t\in I
  \end{equation*}
  in \textbf{exactly one way}.
\end{theorem}
\begin{remark}
  The \nameref{thm: Principle of Superposition} indicates
  the solutions form a vector space, thus there exist bases.
  Therefore, the essence of this theorem is to show that the dimension
  of the solution space is exactly \(n\).
\end{remark}
\begin{proof}
  Let \(\vec{\phi}(t)\) be any solution to the homogeneous system
  and let \(\mat{Y} :=
    \begin{bmatrix}
      \vec{y}_1 & \vec{y}_2 & \cdots & \vec{y}_n
  \end{bmatrix}\).
  We pick some \(t_0 \in I\) and define the constant vector
  \(\vec{\xi} := \vec{\phi}(t_0) \in \mathbb{R}^n\).
  Since \(\vec{y}_1(t), \ldots, \vec{y}_n(t)\) are
  linearly independent for any \(t \in I\),
  we have \(\det(\mat{Y})(t_0) \neq 0\).
  Then the following equation has a unique solution
  \(\vec{c} \in \mathbb{R}^n\):
  \begin{equation*}
    \mat{Y}(t_0) \cdot \vec{c} = \vec{\xi} \qfor \vec{c} \in \mathbb{R}^n.
  \end{equation*}
  Now we define the linear combination of
  \(\vec{y}_1, \vec{y}_2, \ldots, \vec{y}_n\)
  as a new vector function:
  \begin{equation*}
    \vec{\eta}(t) := \mat{Y}(t) \cdot \vec{c} \qfor t\in I.
  \end{equation*}
  Construct the following IVP\@:
  the homogeneous system~(\ref{eq: 1.HL.Sys})
  with the initial condition \(\vec{y}(t_0) = \vec{\xi}\).
  Since \(\vec{\phi}(t)\) and \(\vec{\eta}(t)\) are both solutions
  to this IVP, by the uniqueness of solutions
  (Theorem~\ref{thm: Existence and Uniqueness for 1st-order linear systems}),
  we have \(\vec{\phi}(t) = \vec{\eta}(t)\) for all \(t \in I\).
  Thus any solution \(\vec{\phi}(t)\) can be expressed as a linear
  combination of the fundamental solutions.
\end{proof}

\begin{definition}[Fundamental Set of Solutions]
  A set of \(n\) \textbf{linearly independent} solutions
  \( \{\vec{y}_1(t), \vec{y}_2(t), \ldots, \vec{y}_n(t)\} \)
  to the homogeneous system~(\ref{eq: 1.HL.Sys})
  is called a
  \textbf{fundamental set of solutions}.
\end{definition}

\begin{definition}[Fundamental Matrix]
  {}\label{def: Fundamental Matrix}
  A matrix function \(\mat{Y}(t) \in {C^1(I)}^{n \times n}\) is called a
  \textbf{fundamental matrix} of
  the homogeneous system~(\ref{eq: 1.HL.Sys})
  if its columns form a fundamental set of solutions.
\end{definition}

For a square matrix \(\mat{Y}\) whose columns are solutions
to the homogeneous system~(\ref{eq: 1.HL.Sys}),
we have
\begin{equation*}
  \det(\mat{Y}) \ne 0
  \quad\iff\quad
  \mat{Y} \text{ is a fundamental matrix}.
\end{equation*}

\begin{theorem}[Existence of Fundamental Matrix]
  {}\label{thm: Existence of Fundamental Matrix}
  For the homogeneous system~(\ref{eq: 1.HL.Sys}),
  there exists at least one fundamental matrix \(\mat{Y}(t)\).
\end{theorem}
\begin{proof}
  We need to show that there exist at least \(n\) linearly
  independent solutions.
  We can obtain the solutions with
  Theorem~\ref{thm: Existence and Uniqueness for 1st-order linear systems}
  by constructing \(n\) IVPs with initial conditions
  at the same point \(t_0\) but different values.

  For the \(i\)-th IVP, we set the initial condition as
  \(\vec{y}(t_0) = \vec{e}_i\),
  where \(\vec{e}_i\) is the \(i\)-th standard basis vector.
  Then \(\mat{Y}(t_0) = \mat{I}_n\).
  Since the Wronskian \(W(\vec{y}_1, \ldots, \vec{y}_n)(t_0) =
  \det(\mat{Y})(t_0) = 1 \neq 0\),
  the solutions are linearly independent throughout \(I\)
  by Theorem~\ref{thm: nonzero Wronskian}.
\end{proof}

\begin{theorem}
  Consider the homogeneous
  system~(\ref{eq: 1.HL.Sys})
  with \(\mat{P} \in C(I, \mathbb{R}^{n \times n})\).
  Suppose \(\vec{u}, \vec{v} \in C(I, \mathbb{R}^n)\)
  are two vector functions such that
  \(\vec{y} = \vec{u} + \iu\vec{v} \in C(I, \mathbb{C}^n)\)
  is a solution to the system.
  Then both \(\vec{u}\) and \(\vec{v}\) are solutions to the system.
\end{theorem}

\section{Homogeneous Systems with Constant Coefficients}

In this section, we will focus on the following system
defined on an open interval \(I \subseteq \mathbb{R}\):
\begin{equation}
  \vec{y}'(t) = \mat{A} \cdot \vec{y}(t) \qfor t\in I,
  \label{eq: 1.HLCC.Sys}
\end{equation}
where \(\mat{A} \in \mathbb{R}^{n \times n}\) is a constant matrix.
We will introduce two methods to solve this system:
\begin{enumerate}
  \item The method of matrix exponentials.
  \item The method of eigenvalues and eigenvectors.
\end{enumerate}

\subsection{Method 1: Matrix Exponentials}

\subsubsection{Matrix Exponentials}

\begin{theorem}
  {}\label{thm: exp of sum of commuting matrices}
  For any square matrices \(\mat{A}, \mat{B} \in \mathbb{R}^{n \times n}\),
  \begin{equation*}
    \mat{A} \mat{B} = \mat{B} \mat{A}
    \quad\implies\quad
    e^{\mat{A} + \mat{B}} = e^{\mat{A}} e^{\mat{B}}.
  \end{equation*}
\end{theorem}
\begin{proof}
  Using the Binomial Theorem.
\end{proof}

\begin{theorem}
  {}\label{thm: inv of mat exp}
  For any square matrix \(\mat{A} \in \mathbb{R}^{n \times n}\),
  we have \(\det(e^{\mat{A}}) \ne 0\),
  and \({(e^{\mat{A}})}^{-1} = e^{-\mat{A}}\).
\end{theorem}
\begin{proof}
  Let \(\mat{A}\) be an arbitrary square matrix.
  Since \(\mat{A}(-\mat{A}) = -\mat{A}^2 = (-\mat{A})\mat{A}\),
  we have
  \(e^{\mat{A}} e^{-\mat{A}} = e^{\mat{A} - \mat{A}} = e^{\mat{O}} = \mat{I}\).
\end{proof}

\begin{theorem}
  {}\label{thm: derivative of mat exp}
  For a constant square matrix \(\mat{A}\):
  \begin{equation*}
    \dv{(e^{\mat{A} t})}{t} = \mat{A} e^{\mat{A} t} = e^{\mat{A} t} \mat{A}
  \end{equation*}
\end{theorem}
\begin{proof}
  \begin{equation*}
    \dv{(e^{\mat{A} t})}{t}
    = \dv{t} \sum_{k=0}^{\infty} \frac{\mat{A}^k t^k}{k!}
    = \mat{A} \sum_{k=1}^{\infty} \frac{\mat{A}^{k-1} t^{k-1}}{(k-1)!}
    = \mat{A} e^{\mat{A} t}
    = \sum_{k=1}^{\infty} \frac{\mat{A}^{k-1} t^{k-1}}{(k-1)!} \cdot \mat{A}
    = e^{\mat{A} t} \mat{A}
  \end{equation*}
\end{proof}

Theorem~\ref{thm: derivative of mat exp} indicates that the columns of
the matrix \(\mat{Y}(t) := e^{\mat{A} t}\) are solutions
to the homogeneous system~(\ref{eq: 1.HLCC.Sys}),
and Theorem~\ref{thm: inv of mat exp} shows its invertibility.
Thus, \(\mat{Y}(t)\) is a \textbf{fundamental matrix}.
Then, the \textbf{general solution} to the system can be expressed as
\begin{equation*}
  \vec{y}(t) = e^{\mat{A} t} \cdot \vec{c} \qfor t\in I,
\end{equation*}
where \(\vec{c} \in \mathbb{R}^n\) is an arbitrary constant vector.

\subsubsection{Computing Matrix Exponentials}

Now, how to compute \(e^{\mat{A} t}\)?

\begin{definition}[Semisimple Matrix]
  A square matrix \(\mat{A} \in \mathbb{R}^{n \times n}\)
  is called \textbf{semisimple}
  if it is \textbf{diagonalizable}, i.e.,
  there exists an invertible matrix \(\mat{P}\) such that
  \(\mat{P}^{-1} \mat{A} \mat{P} = \mat{D}\),
  where \(\mat{D}\) is a diagonal matrix.
\end{definition}
\begin{remark}
  The diagonal entries of \(\mat{D}\) are the eigenvalues of \(\mat{A}\).
\end{remark}

\begin{definition}[Nilpotent Matrix]
  A square matrix \(\mat{A} \in \mathbb{R}^{n \times n}\)
  is called \textbf{nilpotent}
  if there exists a positive integer \(k\) such that
  \(\mat{A}^k = \mat{O}\).
\end{definition}

\begin{theorem}
  {}\label{thm: exp of a diagonalizable matrix}
  For a diagonalizable matrix
  \(\mat{A} = \mat{P} \mat{\Lambda} \mat{P}^{-1} \in \mathbb{R}^{n \times n}\),
  we have
  \begin{equation*}
    e^{\mat{A}} = \mat{P} e^{\mat{\Lambda}} \mat{P}^{-1}
    = \mat{P}
    \begin{pmatrix}
      e^{\lambda_1} & & \\ & \ddots & \\ & & e^{\lambda_n}
    \end{pmatrix}
    \mat{P}^{-1}
    \qquad\text{where }
    \lambda_1, \ldots, \lambda_n \text{ are eigenvalues of } \mat{A}.
  \end{equation*}
\end{theorem}

\begin{theorem}[S-N Decomposition]
  {}\label{thm: S-N Decomposition}
  For any square matrix \(\mat{A} \in \mathbb{R}^{n \times n}\),
  there \textbf{uniquely} exist a semisimple matrix \(\mat{S}\)
  and a nilpotent matrix \(\mat{N}\) such that
  \begin{equation*}
    \mat{A} = \mat{S} + \mat{N} \quad\text{and}\quad
    \mat{S} \mat{N} = \mat{N} \mat{S}.
  \end{equation*}
  Additionally, \(\mat{S}\) shares the same eigenvalues as \(\mat{A}\)
  with the same algebraic multiplicities.
\end{theorem}

When computing \(e^{\mat{A} t}\),
there are two cases:
\(\mat{A}\) is diagonalizable or not.
If \(\mat{A}\) is diagonalizable, we can compute
\(e^{\mat{A} t}\) with
Theorem~\ref{thm: exp of a diagonalizable matrix}.
If \(\mat{A}\) is not diagonalizable,
we then apply the \nameref{thm: S-N Decomposition}
to \(\mat{A}\).
Since \(\mat{S}\) and \(\mat{N}\) commute, we have
\begin{equation*}
  e^{\mat{A} t} = e^{\mat{S} t + \mat{N} t}
  = e^{\mat{S} t} e^{\mat{N} t}.
\end{equation*}
Because \(\mat{S}\) is semisimple, we can compute \(e^{\mat{S} t}\)
with Theorem~\ref{thm: exp of a diagonalizable matrix}.
For the nilpotent matrix \(\mat{N}\),
since \(\mat{N}^k = \mat{O}\) for some \(k\),
the series expansion of \(e^{\mat{N} t}\) terminates within dinitely many terms:
\begin{equation*}
  e^{\mat{N} t} = \mat{I} + \mat{N} t + \frac{\mat{N}^2 t^2}{2!} + \cdots
  + \frac{\mat{N}^{k-1} t^{k-1}}{(k-1)!}.
\end{equation*}
Thus it is computable.

\subsection{Method 2: Eigenvalues and Eigenvectors}

\begin{definition}[Generalized Eigenvector]
  {}\label{def: Generalized Eigenvector}
  For a square matrix \(\mat{A} \in \mathbb{R}^{n \times n}\) and an eigenvalue
  \(\lambda \) of \(\mat{A}\), a nonzero vector \(\vec{\xi} \in \mathbb{R}^n\)
  is called a \textbf{generalized eigenvector} of \(\mat{A}\) corresponding to
  \(\lambda \) if there exists a positive integer \(k\) such that
  \begin{equation*}
    {(\mat{A} - \lambda \mat{I})}^k \vec{\xi} = \vec{0}.
  \end{equation*}
  If \(k\) is the least integer satisfying the above equation,
  then \(\vec{\xi}\) is called
  a generalized eigenvector of order \(k\).
\end{definition}

\begin{theorem}[Generalized Eigenspace]
  {}\label{thm: Generalized Eigenspace}
  The \textbf{generalized eigenspace} of
  an eigenvalue \( \lambda \) of \(\mat{A}\) with algebraic multiplicity \(m\),
  denoted by \(G_{\lambda}\), is given by
  \begin{equation*}
    G_{\lambda} = \ker \! \qty({(\mat{A} - \lambda \mat{I})}^m)
    \qand
    \dim(G_{\lambda}) = m.
  \end{equation*}
\end{theorem}


Consider the homogeneous system~(\ref{eq: 1.HLCC.Sys}).
Suppose \(\mat{A} \in \mathbb{R}^{n \times n}\) has \(k\) distinct eigenvalues:
\begin{equation*}
  \det(x\mat{I} - \mat{A}) =
  {(x - \lambda_1)}^{m_1} {(x - \lambda_2)}^{m_2}
  \cdots {(x - \lambda_k)}^{m_k},
\end{equation*}
where \(m_i\) is the algebraic multiplicity of \(\lambda_i\) and
\(\sum_{i=1}^k m_i = n\).
For each eigenvalue \(\lambda_i\), we need to find \(m_i\) linearly
independent solutions:
\( \vec{y}^{(i)}_1, \vec{y}^{(i)}_2, \ldots, \vec{y}^{(i)}_{m_i} \),
so that we have \(n\) linearly independent solutions in total.

For each \(\lambda_i\), whose geometric multiplicity is \(q_i\),
there are two cases:
\begin{enumerate}
  \item \(1 \leqslant q_i = m_i \leqslant n\),
    there are \(m_i\) L.I. eigenvectors;
  \item \(1 \leqslant q_i < m_i \leqslant n\).
\end{enumerate}

\subsubsection*{Case 1}

For the first case, we can construct the solutions for \(\lambda_i\) as
\begin{align*}
  \vec{y}^{(i)}_1(t) &= \vec{r}^{(i)}_1 e^{\lambda_i t}, \\
  \vec{y}^{(i)}_2(t) &= \vec{r}^{(i)}_2 e^{\lambda_i t}, \\
  &\vdots \\
  \vec{y}^{(i)}_{m_i}(t) &= \vec{r}^{(i)}_{m_i} e^{\lambda_i t}.
\end{align*}
where \(\vec{r}^{(i)}_1, \vec{r}^{(i)}_2, \ldots, \vec{r}^{(i)}_{m_i}\)
are \(m_i\) L.I. eigenvectors
corresponding to \(\lambda_i\).

\subsubsection*{Case 2}

By Theorem~\ref{thm: Generalized Eigenspace},
there are \(m_i\) L.I. generalized eigenvectors for \(\lambda_i\):
\(\vec{\xi}^{(i)}_1, \vec{\xi}^{(i)}_2, \ldots, \vec{\xi}^{(i)}_{m_i}\).
We use each of them to construct a solution.
For the \(j\)-th generalized eigenvector \(\vec{\xi}^{(i)}_j\),
we first construct the following \(m_i\) vectors:
\begin{align*}
  \vec{\eta}^{(i,j)}_0 &:= \vec{\xi}^{(i)}_j, \\
  \vec{\eta}^{(i,j)}_1 &:= (\mat{A} - \lambda_i \mat{I}) \vec{\xi}^{(i)}_j, \\
  \vec{\eta}^{(i,j)}_2 &:= {(\mat{A} - \lambda_i \mat{I})}^2
  \vec{\xi}^{(i)}_j, \\
  &\vdots \\
  \vec{\eta}^{(i,j)}_{m_i-1} &:= {(\mat{A} - \lambda_i \mat{I})}^{m_i-1}
  \vec{\xi}^{(i)}_j.
\end{align*}
\begin{remark}
  Some of these vectors may be zero. If \(\vec{\xi}^{(i)}_j\)
  is of order \(s\) (\(1 \leqslant s \leqslant m_i\)),
  then \(\vec{\eta}^{(i,j)}_s = \vec{\eta}^{(i,j)}_{s+1} = \cdots = \vec{0}\),
  and \(\vec{\eta}^{(i,j)}_{s-1}\) is an ordinary eigenvector, while
  the rest \(s-1\) nonzero vectors are generalized eigenvectors (nonzero).
\end{remark}

Now we construct the solution as:
\begin{equation*}
  \vec{y}^{(i)}_j(t) = \qty(
    \vec{\eta}^{(i,j)}_0 +
    \vec{\eta}^{(i,j)}_1 t +
    \vec{\eta}^{(i,j)}_2 \frac{t^2}{2!} + \cdots +
    \vec{\eta}^{(i,j)}_{m_i-1} \frac{t^{m_i-1}}{(m_i-1)!}
  ) \cdot e^{\lambda_i t}
  \quad\qfor j = 1, 2, \ldots, m_i.
\end{equation*}
It can be verified that \(\vec{y}^{(i)}_j\) is indeed a
solution to the system by direct substitution:
\(\displaystyle \dv{t}(\vec{y}^{(i)}_j)=\mat{A} \cdot \vec{y}^{(i)}_j\).

\subsubsection*{The General Solution}

Finally, we can express the general solution to the system as
\begin{align*}
  \vec{y}(t) &= \sum_{i=1}^k \sum_{j=1}^{m_i} c^{(i)}_j \vec{y}^{(i)}_j(t) \\
  &= \sum_{i=1}^k \sum_{j=1}^{m_i} c^{(i)}_j
  \qty(
    \vec{\eta}^{(i,j)}_0 +
    \vec{\eta}^{(i,j)}_1 t +
    \vec{\eta}^{(i,j)}_2 \frac{t^2}{2!} + \cdots +
    \vec{\eta}^{(i,j)}_{m_i-1} \frac{t^{m_i-1}}{(m_i-1)!}
  ) \cdot e^{\lambda_i t} \\
  &= \sum_{i=1}^k \sum_{j=1}^{m_i} c^{(i)}_j e^{\lambda_i t}
  \sum_{s=0}^{m_i-1} \vec{\eta}^{(i,j)}_s \frac{t^s}{s!}
\end{align*}
where \(k\) is the number of distinct eigenvalues,
\(m_i\) is the algebraic multiplicity of \(\lambda_i\),
and \(c^{(i)}_j\) are arbitrary constants.

\paragraph{Relation with the Matrix Exponential}
The above solution can be rewritten as
\begin{align*}
  \vec{y}(t)
  &= \sum_{i=1}^k \sum_{j=1}^{m_i} c^{(i)}_j e^{\lambda_i t}
  \sum_{s=0}^{m_i-1} \vec{\eta}^{(i,j)}_s \frac{t^s}{s!} \\
  &= \sum_{i=1}^k \sum_{j=1}^{m_i}
  c^{(i)}_j e^{\lambda_i t}
  \qty[ \sum_{s=0}^{m_i-1}
    {(\mat{A} - \lambda_i \mat{I})}^{s} \frac{t^s}{s!}
  ] \vec{\xi}^{(i)}_j \\
  &= \sum_{i=1}^k \sum_{j=1}^{m_i}
  c^{(i)}_j e^{\lambda_i t}
  \qty[ \sum_{s=0}^{\infty}
    {(\mat{A} - \lambda_i \mat{I})}^{s} \frac{t^s}{s!}
  ] \vec{\xi}^{(i)}_j \\
  &= e^{\mat{A} t} \sum_{i=1}^k \sum_{j=1}^{m_i}
  c^{(i)}_j \vec{\xi}^{(i)}_j
\end{align*}
Note that for each eigenvalue \(\lambda_i\),
\(\vec{\xi}^{(i)}_1, \vec{\xi}^{(i)}_2, \ldots, \vec{\xi}^{(i)}_{m_i}\)
are \(m_i\) L.I. generalized eigenvectors we picked,
whose existence is guaranteed by Theorem~\ref{thm: Generalized Eigenspace}.
Additionally, generalized eigenvectors corresponding to different eigenvalues
are also linearly independent.
Thus, the span of all these \(n\) generalized eigenvectors is
exactly the whole space \(\mathbb{R}^n\).
We can rewrite the linear combination of them into
a single arbitrary constant vector \(\vec{c} \in \mathbb{R}^n\),
which gives the same result as the matrix exponential method.
