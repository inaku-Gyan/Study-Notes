\part*{Review of Matrices}

\section{}


\begin{definition}[Eigenvalues and Eigenvectors]
  For a square matrix \(\mat{A}\), if there exists a non-zero vector \(\vec{v}\)
  and a scalar \(\lambda \) such that
  \[
    \mat{A}\vec{v} = \lambda\vec{v},
  \]
  then \(\lambda \) is called an eigenvalue of \(\mat{A}\) and
  \(\vec{v}\) is called an eigenvector corresponding to \(\lambda \).
\end{definition}

\begin{definition}[Multiplicity of Eigenvalues]
  Suppose \(\lambda \) is an eigenvalue of a matrix \(\mat{A}\) and let the
  characteristic polynomial of \(\mat{A}\) be \(p(x) = \det(\mat{A} - x \mat{I})\).
  Then we have two types of multiplicity for \(\lambda \):
  \begin{itemize}
    \item \textbf{Algebraic multiplicity} of \(\lambda \):
      The number of times \(\lambda \) appears as a root of \(p(x)\).
    \item \textbf{Geometric multiplicity} of \(\lambda \):
      It is equivalent to any of the following (all of which are equal):
      \begin{itemize}
        \item The number of linearly independent eigenvectors
          corresponding to \(\lambda \).
        \item The dimension of the corresponding eigenspace:
          \( \dim(\{\vec{v} \mid \mat{A}\vec{v} = \lambda\vec{v}\}) \).
        \item The nullity of \(\mat{A} - \lambda \mat{I}\):
          \( \dim(\Null(\mat{A} - \lambda \mat{I})) \).
      \end{itemize}
  \end{itemize}
\end{definition}

\begin{theorem}
  For any eigenvalue \(\lambda \) of a square matrix \(\mat{A}\),
  \begin{equation*}
    \text{geometric multiplicity of } \lambda
    \quad \leqslant \quad
    \text{algebraic multiplicity of } \lambda.
  \end{equation*}
\end{theorem}
\begin{proof}
  Suppose \(\mat{A} \in \mathbb{R}^{n \times n}\) and \(\lambda \) is
  an eigenvalue of \(\mat{A}\) with geometric multiplicity \(k\).
  Since \(\dim(\Null(\mat{A} - \lambda \mat{I})) = k\),
  we can pick \(k\) orthonormal eigenvectors
  \( \{\vec{v}_1, \vec{v}_2, \ldots, \vec{v}_k\} \) from \(\Null(\mat{A} - \lambda \mat{I})\),
  and then extend this set to an orthonormal basis of \(\mathbb{R}^n\)
  by adding \(n-k\) more vectors
  \( \{\vec{v}_{k+1}, \vec{v}_{k+2}, \ldots, \vec{v}_n\} \).
  Define the following matrices:
  \begin{equation*}
    \mat{V}_1 =
    \begin{bmatrix}
      \vrule & & \vrule \\
      \vec{v}_1 & \cdots & \vec{v}_k \\
      \vrule & & \vrule
    \end{bmatrix}
    \qquad
    \mat{V}_2 =
    \begin{bmatrix}
      \vrule & & \vrule \\
      \vec{v}_{k+1} & \cdots & \vec{v}_n \\
      \vrule & & \vrule
    \end{bmatrix}
    \qquad
    \mat{V} =
    \begin{bmatrix}
      & & & & \\
      & \ldots \mat{V}_1 & & \ldots \mat{V}_2 & \\
      & & & &
    \end{bmatrix} =
    \begin{bmatrix}
      \vrule & & \vrule \\
      \vec{v}_{1} & \cdots & \vec{v}_n \\
      \vrule & & \vrule
    \end{bmatrix}
  \end{equation*}
  where
  \(\mat{V}_1 \in \mathbb{R}^{n \times k} \) and
  \(\mat{V}_2 \in \mathbb{R}^{n \times (n-k)}\),
  and \(\mat{V} \in \mathbb{R}^{n \times n}\) is an orthogonal matrix.
  Then we have
  \begin{align*}
    \mat{A} \mat{V}_1 = \lambda \mat{V}_1,
    \qquad\qquad
    \mat{V}^\top \mat{A} \mat{V} &=
    \begin{bmatrix}
      & & \\
      & \ldots \mat{V}_1^\top & \\
      & & \\
      & \ldots \mat{V}_2^\top & \\
      & &
    \end{bmatrix} \mat{A}
    \begin{bmatrix}
      & & & & \\
      & \ldots \mat{V}_1 & & \ldots \mat{V}_2 & \\
      & & & &
    \end{bmatrix} \\
    &=
    \begin{bmatrix}
      & & \\
      & \ldots (\mat{V}_1^\top \mat{A} \mat{V}_1) & \ldots (\mat{V}_1^\top \mat{A} \mat{V}_2) & \\
      & & \\
      & \ldots (\mat{V}_2^\top \mat{A} \mat{V}_1) & \ldots (\mat{V}_2^\top \mat{A} \mat{V}_2) & \\
      & &
    \end{bmatrix} \\
    &=
    \begin{bmatrix}
      & & \\
      & \ldots (\lambda \mat{I}_k) & \ldots (\mat{V}_1^\top \mat{A} \mat{V}_2) & \\
      & & \\
      & \ldots \mat{O}_{n-k} & \ldots (\mat{V}_2^\top \mat{A} \mat{V}_2) & \\
      & &
    \end{bmatrix}
    \in \mathbb{R}^{n \times n}
  \end{align*}
  Thus, the characteristic polynomial of \(\mat{A}\) can be computed as
  \begin{align*}
    p(x) =
    \det(\mat{A} - x \mat{I})
    &= \det(\mat{V}^\top (\mat{A} - x \mat{I}) \mat{V} ) \\
    &= \det(\mat{V}^\top \mat{A} \mat{V} - x \mat{I}) \\
    &= \det\left(
      \begin{bmatrix}
        & & \\
        & \ldots (\lambda \mat{I}_k - x \mat{I}_k) & \ldots (\mat{V}_1^\top \mat{A} \mat{V}_2) & \\
        & & \\
        & \ldots \mat{O}_{n-k} & \ldots (\mat{V}_2^\top \mat{A} \mat{V}_2 - x \mat{I}_{n-k}) & \\
        & &
      \end{bmatrix}
    \right) \\
    &= {(\lambda - x)}^k
    \cdot
    \det(\mat{V}_2^\top \mat{A} \mat{V}_2 - x \mat{I}_{n-k})
  \end{align*}
  Therefore, \(\lambda \) is a root of \(p(x)\) with multiplicity at
  least \(k\),
  which means the algebraic multiplicity of \(\lambda \) is at least \(k\).
\end{proof}
