\part*{First Order ODEs}

The \textbf{general form} of a first order ODE is
\[
  \dv{y}{t} = f(t,y)
\]
for some given function \(f\).

\section{Linear}

The \textbf{general form} of a first order linear ODE is
\[
  \dv{y}{t} + p(t)\cdot y = q(t)
\]
for some given functions \(p\) and \(q\).

\subsection{Method of Integrating Factors}

\begin{theoremBox}
  For ODEs of the form
  \[
    \dv{y}{t} + p(t)\cdot y = q(t)
  \]
  the \textbf{general solution} is
  \[
    y = \frac{1}{\mu(t)} \int \mu(t)\cdot q(t)\dd{t}
  \]
  where the \textbf{integrating factor} is any \(\mu(t)\) that satisfies
  \[
    \ln\abs{\mu(t)} = \int p(t)\dd{t}
  \]
\end{theoremBox}
\begin{remark}
  Indefinite integrals yield a family of functions differing by a constant.
  Each indefinite integral results in an independent constant, which
  represents a degree of freedom.
  However, the constant from \(\mu(t)\) cancels out in the final
  solution for \(y\).
\end{remark}


\subsection{Variation of Parameters}

\begin{theoremBox}
  For ODEs of the form
  \[
    \dv{y}{t} + p(t)\cdot y = q(t)
  \]
  let \(h(t)\) be a function such that
  \[
    h(t) = \int p(t)\dd{t}
  \]
  Then the \textbf{general solution} is given by
  \[
    y = e^{-h(t)} \int q(t)\cdot e^{h(t)} \dd{t}
  \]
\end{theoremBox}

\subsection{}

\begin{theoremBox}
  For ODEs of the form
  \[
    \dv{y}{t} + p(t)\cdot y = q(t)
  \]
  the \textbf{general solution} is given by
  \[
    y = v \int \frac{q(t)}{v} \dd{t}
  \]
  where \(v\) satisfies
  \[
    \dv{v}{t} + p(t)\cdot v = 0
  \]
\end{theoremBox}
\begin{proof}
  We want to solve \cref{eq: method3 eq for y}.
  Suppose \(v\) satisfies \cref{eq: method3 eq for v},
  which is separable and can be solved easily.
  \begin{gather}
    y' + p(t)\cdot y = q(t)
    \label{eq: method3 eq for y} \\
    v' + p(t)\cdot v = 0
    \label{eq: method3 eq for v}
  \end{gather}
  Take the derivative of \(y/v\):
  \begin{align*}
    \dv{t}(\frac{y}{v})
    &= \frac{y' v - v' y}{v^2} \\
    &= \frac{y' v + (p(t)\cdot v) y}{v^2}
    \tag*{Plug in \cref{eq: method3 eq for v}.} \\
    &= \frac{q(t)}{v}
    \tag*{Plug in \cref{eq: method3 eq for y}.}
  \end{align*}
  Thus we have
  \begin{equation*}
    \frac{y}{v} = \int \frac{q(t)}{v} \dd{t}
  \end{equation*}
  Since \(v\) can be solved from \cref{eq: method3 eq for v},
  we obtain the solution for \(y\).
\end{proof}

\section{Non-Linear}

\subsection{Separable Equations}

A first order ODE is \textbf{separable} if it can be written in the form
\[
  M(t) + N(y)\cdot \dv{y}{t} = 0
\]
for some functions \(M\) and \(N\).
Then the solution satisfies
\[
  \int N(y)\dd{y} = -\int M(t)\dd{t}
\]

\subsection{Transformation Methods}

Some complicated non-linear ODEs can be transformed into ODEs that
are solvable by suitable substitutions.

\subsubsection{Bernoulli Equations}

Bernoulli equations can be transformed into linear ODEs by the
substitution \(v = y^{1-r}\).


\begin{definition}[Bernoulli Equations]
  A first order ODE is called a \textbf{Bernoulli equation} if it can
  be written in the form
  \begin{equation}
    \dv{y}{t} + p(t)\cdot y = q(t)\cdot y^r
    \label{eq: Bernoulli Equations}
  \end{equation}
  for some given functions \(p\), \(q\) and
  a real number \( r\in\mathbb{R} \setminus \{ 0, 1 \} \).
\end{definition}

\begin{theoremBox}
  The \textbf{general solution} of a Bernoulli equation~\eqref{eq:
  Bernoulli Equations} is given by
  \[
    y^{1-r} = \frac{1-r}{\mu(t)} \int \mu(t)\cdot q(t)\dd{t}
  \]
  where \(\mu(t)\) satisfies
  \[
    \ln\abs{\mu(t)} = (1-r)\int p(t)\dd{t}
  \]

  Note: If \(r > 0\), then \(y = 0\) is also a solution,
  which is not captured by the above formula.
\end{theoremBox}

\begin{proof}
  Substitute \(v = y^{1-r}\), and then solve with the method of
  integrating factors.
  \begin{align*}
    \dv{y}{t} + p(t)\cdot y = q(t)\cdot y^r
    \quad & \implies\quad
    y^{-r}\dv{y}{t} + p(t)\cdot y^{1-r} = q(t)
    \qfor y \neq 0 \\
    \quad & \implies\quad
    \dv{t}( y^{1-r}) + (1-r)p(t)\cdot y^{1-r} = (1-r) q(t)
  \end{align*}
\end{proof}

\subsubsection{Homogeneous Equations}

Homogeneous equations can be transformed into separable equations.

\begin{theoremBox}
  If a first-order ODE can be expressed in the following form, then
  it is said to be \textbf{homogeneous}:
  \[
    \dv{y}{t} = F(y/t)
  \]
  for some given function \(F\).

  \noindent
  Then it can be transformed into a separable equation using \(v = y/t\):
  \[
    \frac{1}{F(v) - v} \dd{v} = \frac{1}{t} \dd{t}
  \]
\end{theoremBox}

\begin{remark}
  The homogeneous equations considered here have nothing to do with
  the homogeneous LCCDEs.
\end{remark}

\subsection{Exact Equations}

\begin{definition}[Exact Equations]
  A first order ODE \(M(t,y)\dd{t}+N(t,y)\dd{y}=0\) is said to be
  \textbf{exact} if there exists a function \(\Psi(t,y)\) such that
  \begin{equation*}
    \pdv{\Psi(t,y)}{t} = M(t,y) \qand \pdv{\Psi(t,y)}{y} = N(t,y)
  \end{equation*}
  Then the general solution is given implicitly as
  \[
    \Psi(t,y) = C \qquad\text{for some constant } C \in \mathbb{R}
  \]
\end{definition}

\begin{theorem}
  Let \(R\) be a simply connected region in the \(t y\)-plane.
  If \(M(t,y)\), \(N(t,y)\), \(M_y(t,y)\), and \(N_t(t,y)\) are
  continuous in~\(R\), then
  \begin{equation*}
    M(t,y) + N(t,y) \dv{y}{t} = 0 \text{ is exact in } R
    \qquad\iff\qquad
    M_y(t,y) = N_t(t,y) \quad \text{for all } (t,y) \in R
  \end{equation*}
  where \(M_y(t,y) := \pdv*{M(t,y)}{y}\) and \(N_t(t,y) := \pdv*{N(t,y)}{t}\).
\end{theorem}

To solve the first order ODE \(M(t,y)\dd{t}+N(t,y)\dd{y}=0\),
we can follow these steps:
\begin{enumerate}
  \item Verify that the equation is exact by checking if \(M_y(t,y) =
    N_t(t,y)\).
  \item Integrate \(M(t,y)\) with respect to \(t\) to find \(\Psi(t,y)\):
    \[
      \Psi(t,y) = \int M(t,y)\dd{t} + h(y)
    \]
    where \(h(y)\) is an arbitrary function of \(y\).
  \item Differentiate \(\Psi(t,y)\) with respect to \(y\) and set it
    equal to \(N(t,y)\):
    \[
      \pdv{\Psi(t,y)}{y} = N(t,y)
    \]
    This will allow us to solve for \(h(y)\).
  \item Substitute \(h(y)\) back into \(\Psi(t,y)\) to obtain the
    implicit solution:
    \[
      \Psi(t,y) = C
    \]
    for some constant \(C\).
\end{enumerate}

\subsection{Exact Equations and Integrating Factors}

It is sometimes possible to convert an ODE that is not exact into an
exact equation with a suitable integrating factor.

Suppose the following first order ODE is exact:
\begin{equation*}
  \mu M(t,y) + \mu N(t,y)\dv{y}{t} = 0
\end{equation*}
Then the integrating factor \( \mu \) must satisfy
\begin{equation*}
  \pdv{(\mu M)}{y} = \pdv{(\mu N)}{t}
\end{equation*}

On one hand, assuming that \( \mu = \mu(t) \), we have
\begin{equation*}
  \dv{\mu}{t} = \frac{M_y - N_t}{N}\cdot \mu
\end{equation*}
If \((M_y - N_t) / N\) depends on \(t\) alone, then we can solve for \(\mu(t)\).

On the other hand, assuming that \( \mu = \mu(y) \), we have
\begin{equation*}
  \dv{\mu}{y} = \frac{N_t - M_y}{M}\cdot \mu
\end{equation*}
If \((N_t - M_y) / M\) depends on \(y\) alone, then we can solve for \(\mu(y)\).

If neither case applies, then try some substitutions to transform the
ODE into a solvable form.

\section{The Existence and Uniqueness Theorem}

\subsection{The Theorem}

\begin{theorem}[Existence and Uniqueness Theorem for First Order Linear ODEs]
  For the IVP\@:
  \begin{equation*}
    \dv{y}{t} = p(t)\cdot y + q(t) \qand y(t_0) = y_0
  \end{equation*}
  if there exists an interval \(I \subseteq \mathbb{R}\) such that
  \(t_0 \in I\) and \(p(t)\), \(q(t)\) are \textbf{continuous} on \(I\),
  then there exists a \textbf{unique} solution \(y(t)\) defined on \(I\)
  that satisfies the IVP\@.
\end{theorem}

\begin{remark}
  The solution is defined on the entire interval \(I\).
\end{remark}

\begin{theorem}[Existence and Uniqueness Theorem]
  {}\label{thm: Existence and Uniqueness Theorem}
  For the IVP\@:
  \begin{equation*}
    \dv{y}{t} = f(t,y) \qand y(t_0) = y_0
  \end{equation*}
  if there exists a closed rectangle
  \(R: t \in [t_0 - a, t_0 + a], y \in [y_0 - b, y_0 + b]\)
  such that \(f(t,y)\) and \(\pdv*{f(t,y)}{y}\)
  are \textbf{continuous} on \(R\),
  then there exists a \textbf{unique} solution \(y = \phi(t)\)
  defined for \(t \in [t_0 - h, t_0 + h]\),
  where \(h\) is some number satisfying
  \(0 < \min \{b/M, a\} \leqslant h \leqslant a \)
  and \(M = \max\limits_{(t,y) \in R}\abs{f(t,y)} < +\infty \).
\end{theorem}

\begin{remark}
  The solution is guaranteed to be defined at least on the interval of \(t\)
  such that \(\abs{t - t_0} \leqslant \min \{b/M, a\} \),
  where \(M\) is the bound of \(f(t,y)\) on the closed rectangle \(R\).
\end{remark}

\begin{remark}
  If the conditions of the theorem are not satisfied,
  then the solution may not be unique or may not exist.
\end{remark}

\begin{theorem}[Existence Theorem]
  For the IVP\@:
  \begin{equation*}
    \dv{y}{t} = f(t,y) \qand y(t_0) = y_0
  \end{equation*}
  if there exists a closed rectangle
  \(R: t \in [t_0 - a, t_0 + a], y \in [y_0 - b, y_0 + b]\)
  such that \(f(t,y)\) is \textbf{continuous} on \(R\),
  then there exists \textbf{at least one} solution \(y = \phi(t)\)
  defined for \(t \in [t_0 - h, t_0 + h]\),
  where \(h\) is some number satisfying
  \(0 < \min \{b/M, a\} \leqslant h \leqslant a \)
  and \(M = \max\limits_{(t,y) \in R}\abs{f(t,y)} < +\infty \).
\end{theorem}
\begin{remark}
  When \(\pdv*{f}{y}\) may not be continuous, the solution may not be unique.
\end{remark}

\subsection{The Preliminary before the Proof}

\begin{definition}[Pointwise Convergence]
  A sequence of functions \( \{f_n\} \) defined on a set
  \(\mathcal{D} \subseteq \mathbb{R}\) is said to
  \textbf{converge pointwise} to a function \(f\) if:
  \begin{equation*}
    \forall x \in \mathcal{D}, \lim_{n\to\infty} f_n(x) = f(x)
  \end{equation*}
  The \textbf{rate of convergence} can be \textbf{different}
  for different points in \(\mathcal{D}\).
\end{definition}

\begin{definition}[Uniform Convergence]
  A sequence of functions \( \{f_n\} \) defined on a set
  \(\mathcal{D} \subseteq \mathbb{R}\) is said to
  \textbf{converge uniformly} to a function \(f\) if:
  \begin{equation*}
    \forall \varepsilon > 0, \exists N \in \mathbb{N}, \forall n > N,
    \forall x \in \mathcal{D}, \abs{f_n(x) - f(x)} < \varepsilon
  \end{equation*}
  The \textbf{rate of convergence} is the \textbf{same} for all points in
  \(\mathcal{D}\).
\end{definition}

\begin{theoremBox}
  Uniform convergence implies pointwise convergence.
\end{theoremBox}
\begin{proof}
  The proof is given in first-order formal logic.
  The inference rules used are the following:
  \begin{itemize}
    \item \(\alpha \to \beta \iff \neg \alpha \lor \beta \)
    \item \(\textcolor{red}{\forall x}(\textcolor{blue}{\alpha} \; \lor P(x))
        \iff \textcolor{blue}{\alpha}\; \lor \; \textcolor{red}{\forall x} P(x)
        \qand \exists x (\textcolor{blue}{\alpha} \; \lor P(x))
        \iff \textcolor{blue}{\alpha}\; \lor \;
      \exists x P(x)\)
    \item \(\textcolor{red}{\forall x} \forall y P(x,y) \iff
        \forall y \textcolor{red}{\forall x} P(x,y)
        \qand \exists x \textcolor{red}{\forall y} P(x,y)
        \textcolor{blue}{\implies}
      \textcolor{red}{\forall y} \exists x P(x,y)\)
  \end{itemize}
  To make it easier,
  we suppose all the variables are restricted to real numbers.
  Rewrite the definition of uniform convergence into first-order formal logic:
  \begin{align*}
    \text{Uniform Convergence}
    &\iff
    \forall \varepsilon \Big(
      \varepsilon > 0 \to \exists N \Big[
        N \in \mathbb{N} \to \forall n \Big(
          n > N \to
          \forall x \big(
            x \in \mathcal{D} \to
            \abs{f_n(x) - f(x)} < \varepsilon
          \big)
        \Big)
      \Big]
    \Big) \\
    &\iff
    \forall \varepsilon \Big(
      \varepsilon \leqslant 0 \; \lor \; \exists N \Big[
        N \notin \mathbb{N} \; \lor \; \forall n \Big(
          n \leqslant N \; \lor \;
          \textcolor{red}{\forall x} \big(
            x \notin \mathcal{D} \; \lor \;
            \abs{f_n(x) - f(x)} < \varepsilon
          \big)
        \Big)
      \Big]
    \Big) \\
    &\textcolor{blue}{\implies}
    \textcolor{red}{\forall x} \forall \varepsilon \Big(
      \varepsilon \leqslant 0 \; \lor \; \exists N \Big[
        N \notin \mathbb{N} \; \lor \; \forall n \Big(
          n \leqslant N \; \lor \;
          \textcolor{blue}{x \notin \mathcal{D}} \; \lor \;
          \abs{f_n(x) - f(x)} < \varepsilon
        \Big)
      \Big]
    \Big) \\
    &\iff
    \forall x \Big[
      \textcolor{blue}{x \notin \mathcal{D}} \; \lor \;
      \forall \varepsilon \Big(
        \varepsilon \leqslant 0 \; \lor \; \exists N \Big[
          N \notin \mathbb{N} \; \lor \; \forall n \Big(
            n \leqslant N \; \lor \;
            \abs{f_n(x) - f(x)} < \varepsilon
          \Big)
        \Big]
      \Big)
    \Big] \\
    &\iff
    \forall x \Big[
      x \in \mathcal{D} \to
      \forall \varepsilon \Big(
        \varepsilon > 0 \to \exists N \Big[
          N \in \mathbb{N} \to \forall n \Big(
            n > N \to
            \abs{f_n(x) - f(x)} < \varepsilon
          \Big)
        \Big]
      \Big)
    \Big] \\
    &\iff
    \forall x \in \mathcal{D}, \forall \varepsilon > 0,
    \exists N \in \mathbb{N}, \forall n > N,
    \abs{f_n(x) - f(x)} < \varepsilon \\
    &\iff \text{Pointwise Convergence}
  \end{align*}
\end{proof}

\begin{theorem}[Weierstrass M-test]\label{thm: Weierstrass M-test}
  Let \( \{\phi_n\} \) be a sequence of functions defined on a set
  \(\mathcal{D} \subseteq \mathbb{R}\).
  If there exists a sequence of positive numbers \( \{M_n\} \) such that
  \begin{equation*}
    \forall n \in \mathbb{N}, \forall x \in \mathcal{D},
    \abs{\phi_n(x)} < M_n
    \qand \sum_{n=1}^{\infty} M_n \text{ converges}
  \end{equation*}
  then the new sequence \( \{\Phi_n(x)\} \),
  where \(\displaystyle \Phi_n(x) := \sum_{k=1}^{n} \phi_k(x)\),
  \textbf{converges uniformly} on \(\mathcal{D}\).
\end{theorem}

\begin{theorem}\label{thm: limit and integration can be interchanged}
  Suppose \( \{\phi_n(x)\} \) is a sequence of functions defined on
  an interval
  \([a,b] \subseteq \mathbb{R}\) and is \textbf{uniformly convergent},
  then the operations of limit and integration can be interchanged:
  \begin{equation*}
    \lim_{n\to\infty} \int_{a}^{b} \phi_n(x) \dd{x} =
    \int_{a}^{b} \lim_{n\to\infty} \phi_n(x) \dd{x}
  \end{equation*}
\end{theorem}

\subsection{The Proof}

In this section, we will use the \textbf{method of successive approximations},
aka \textbf{Picard's iteration method},
to prove Theorem~\ref{thm: Existence and Uniqueness Theorem}.
Suppose the IVP is given by \( \dv*{y}{t} = f(t,y) \) and \( y(t_0) = y_0 \).
And both the functions \(f\) and \(\pdv*{f}{y}\) are continuous on the rectangle
\(R: t \in [t_0 - a, t_0 + a], y \in [y_0 - b, y_0 + b]\).

We start by defining a sequence of functions \( \{\phi_n\} \) as follows:
\begin{gather*}
  \phi_0(t) = y_0 \\
  \phi_{n+1}(t) = y_0 + \int_{t_0}^{t} f(\tau, \phi_n(\tau)) \dd{\tau}
  \qquad\text{for } n = 0, 1, 2, \dots
\end{gather*}
Note that each \(\phi_n\) satisfies the initial condition \(
\phi_n(t_0) = y_0 \).
Then we will take the following steps to complete the proof:
\begin{enumerate}
  \item Find the interval of \(t\) on which all \(\phi_n\) are
    well-defined and continuous.
  \item Show that \( \{ \phi_n(t) \} \) is uniformly convergent on
    that interval.
  \item Show that the limit function
    \( \phi(t) := \lim\limits_{n\to\infty} \phi_n(t) \)
    is a solution to the IVP\@.
  \item Show that the solution is unique.
\end{enumerate}

\subsubsection*{Step 1. The Solution Interval}

Since \(f(t,y)\) is continuous on the rectangle
\(R: [t_0 - a, t_0 + a] \times [y_0 - b, y_0 + b]\),
we need to find an interval of \(t\)
such that \((t, \phi_n(t))\) is in \(R\) for all \(n\),
so that all \(\phi_n(t)\) are well-defined and continuous on that interval.
The result is \([t_0 - h, t_0 + h]\),
where \(h := \min \{b/M, a\} \) and
\(M := \max\limits_{(t,y) \in R} \abs{f(t,y)} \).
We can verify this by mathematical induction.

First, it is clear that \(\phi_0(t) = y_0\) is continuous and
is restricted to \(R\) when \(t \in [t_0 - h, t_0 + h]\).
Now suppose the conclusion holds for \(\phi_k(t)\).
Then we have
\begin{align*}
  \abs{\phi_{k+1}(t) - y_0}
  &= \abs{y_0 + \int_{t_0}^{t} f(\tau, \phi_k(\tau)) \dd{\tau} - y_0} \\
  &= \abs{\int_{t_0}^{t} f(\tau, \phi_k(\tau)) \dd{\tau}} \\
  &\leqslant \int_{t_0}^{t} \abs{f(\tau, \phi_k(\tau))} \dd{\tau}
\end{align*}
Since \(f(t,y)\) is continuous on \(R\) and
\((\tau, \phi_k(\tau))\) is in \(R\) for all \(\tau \) in the integral,
\(f(\tau, \phi_k(\tau))\) is bounded, thus
\begin{align*}
  \abs{\phi_{k+1}(t) - y_0}
  &\leqslant \int_{t_0}^{t} \abs{f(\tau, \phi_k(\tau))} \dd{\tau} \\
  &\leqslant \int_{t_0}^{t} M \dd{\tau}
  \qq{where} M := \max_{(t,y) \in R} \abs{f(t,y)} \\
  &= M|t - t_0| \leqslant M h \leqslant b
\end{align*}
which implies that \(\phi_{k+1}(t) \in [y_0 - b, y_0 + b]\)
for all \(t \in [t_0 - h, t_0 + h]\).
And since \(f(\tau, \phi_k(\tau))\) is continuous,
the integral also results in a continuous function.
Therefore, by induction, we have proven that the sequence
\( \{\phi_n(t)\} \) is well-defined and continuous on \([t_0 - h, t_0 + h]\).

\subsubsection*{Step 2. Uniform Convergence}

We show that the sequence \( \{\phi_n(t)\} \) is uniformly convergent
using Theorem~\ref{thm: Weierstrass M-test}, the Weierstrass M-test.

Define \(g_n(t) := \phi_n(t) - \phi_{n-1}(t)\) for \(n = 1, 2, \dots \),
so that \(\displaystyle\phi_n(t) = \phi_0(t) + \sum_{k=1}^{n} g_k(t)\).
Then we need to find a sequence of positive numbers \( \{M_n\} \) such that
\begin{equation*}
  \forall t \in [t_0 - h, t_0 + h], \abs{g_n(t)} < M_n
  \qand \sum_{n=1}^{\infty} M_n \text{ converges.}
\end{equation*}

Since \(\displaystyle \pdv{f}{y}\) is continuous on \(R\),
for any \((t,y_1), (t,y_2) \in R\),
there exists \(y_3\) between \(y_1\) and \(y_2\) such that
\begin{equation*}
  \frac{f(t,y_1) - f(t,y_2)}{y_1 - y_2} =
  \pdv{f(t,y)}{y}\Big|_{y=y_3}
\end{equation*}
Since \(\displaystyle \pdv{f}{y}\) is also bounded on \(R\),
there exists \(K > 0\) such that
\begin{equation*}
  \abs{f(t,y_1) - f(t,y_2)} \leqslant K \abs{y_1 - y_2}
\end{equation*}
Thus we have
\begin{align*}
  \abs{g_{n+1}(t)}
  &= \abs{\phi_{n+1}(t) - \phi_n(t)} \\
  &\leqslant \int_{t_0}^{t} \abs{ \vphantom{\frac{}{}}
  f(\tau, \phi_n(\tau)) - f(\tau, \phi_{n-1}(\tau))} \dd{\tau} \\
  &\leqslant K \int_{t_0}^{t} \abs{g_n(\tau)} \dd{\tau}
  \intertext{Let \(\displaystyle M := \max\limits_{(t,y) \in R} \abs{f(t,y)}\).}
  \abs{g_{1}(t)}
  &\leqslant \int_{t_0}^{t} \abs{f(\tau, y_0)} \dd{\tau} \\
  &\leqslant M|t - t_0|
  \intertext{By mathematical induction, we can prove that}
  \abs{g_{n}(t)}
  &\leqslant \frac{M K^{n-1} |t - t_0|^n}{n!}
  \intertext{Then, let
  \(\displaystyle M_n := \frac{M K^{n-1} h^n}{n!}\), we have}
  \abs{g_{n}(t)} &\leqslant M_n \qquad \forall t \in [t_0 - h, t_0 + h] \\
  \sum_{n=1}^{\infty} M_n
  &= \frac{M}{K} \sum_{n=1}^{\infty} \frac{{(K h)}^n}{n!} \\
  &= \frac{M}{K} (e^{K h} - 1)
  \qq{converges}
\end{align*}

Therefore, by Theorem~\ref{thm: Weierstrass M-test},
the sequence \( \{\phi_n(t)\} \) is uniformly convergent
on \([t_0 - h, t_0 + h]\).

\subsubsection*{Step 3. The Existence of the Solution}

Let \(\displaystyle \phi(t) := \lim_{n\to\infty} \phi_n(t)\). Then
\begin{align*}
  &
  \phi_{n+1}(t) = y_0 + \int_{t_0}^{t} f(\tau, \phi_n(\tau)) \dd{\tau} \\
  &\implies \lim_{n\to\infty} \phi_{n+1}(t) =
  y_0 + \lim_{n\to\infty} \int_{t_0}^{t} f(\tau, \phi_n(\tau)) \dd{\tau} \\
  &\implies \lim_{n\to\infty} \phi_{n+1}(t) =
  y_0 + \int_{t_0}^{t} \lim_{n\to\infty}  f(\tau, \phi_n(\tau)) \dd{\tau}
  \qquad\qq{by Theorem~\ref{thm: limit and integration can be interchanged}} \\
  &\implies \phi(t) = y_0 + \int_{t_0}^{t} f(\tau, \phi(\tau)) \dd{\tau}
  \qquad\qq{as \(f(t,y)\) is continuous} \\
\end{align*}
Therefore, \(\phi(t)\) is a solution to the IVP\@.

\subsubsection*{Step 4. The Uniqueness of the Solution}

Suppose \(\psi(t)\) is another solution to the IVP\@.
Then we have
\begin{align*}
  \abs{\phi(t) - \psi(t)}
  &= \abs{\int_{t_0}^{t} f(\tau, \phi(\tau)) - f(\tau, \psi(\tau)) \dd{\tau}} \\
  &\leqslant K \int_{t_0}^{t} \abs{\phi(\tau) - \psi(\tau)} \dd{\tau}
  \qq{since \(\displaystyle\pdv{f}{y}\) is continuous and bounded on \(R\)}
\end{align*}
Let \(\Psi(t) := \abs{\phi(t) - \psi(t)}\).
\begin{align*}
  &\dv{t} \Psi(t) \leqslant K \Psi(t)
  \qand \Psi(t_0) = 0 \\
  &\implies e^{-K t} \left(\dv{t}\Psi(t) - K \Psi(t)\right) \leqslant 0 \\
  &\implies \dv{t}(e^{-K t} \Psi(t)) \leqslant 0 \\
  &\implies \int_{t_0}^{t} \dv{\tau}(e^{-K \tau} \Psi(\tau))
  \dd{\tau} \leqslant 0 \\
  &\implies e^{-K t} \Psi(t) \leqslant e^{-K t_0} \Psi(t_0) = 0 \\
  &\implies \Psi(t) = 0 \qquad \forall t \in [t_0 - h, t_0 + h] \\
  &\implies \phi(t) = \psi(t) \qquad \forall t \in [t_0 - h, t_0 + h]
\end{align*}
Therefore, the solution to the IVP\@ is unique.
