\section{Properties of Convolution}

\begin{align*}
    \abs{n} = n \left(u[n-1] - u[-n]\right) = n \left(u[n] - u[-n-1]\right) \\
    f[\abs{n}] = f[n] u[n-1] + f[-n] u[-n] = f[n] u[n] + f[-n] u[-n-1]
\end{align*}
